using static System.Windows.Forms.VisualStyles.VisualStyleElement;

namespace username_password
{
    public partial class Form1 : Form
    {
        public Form1()
        {
            InitializeComponent();

            // Set the form's default button to trigger when the Enter key is pressed
            // Enter tuşuna basıldığında tetiklenecek varsayılan butonu belirliyoruz
            this.AcceptButton = button1;
        }

        private void button1_Click(object sender, EventArgs e)
        {
            // Capture the current input values from the text boxes
            // Metin kutularındaki anlık girdi değerlerini alıyoruz
            string username = textBox1.Text;
            string password = textBox2.Text;

            // Validate credentials for authorized users
            // Yetkili kullanıcılar için kimlik bilgilerini doğruluyoruz
            if ((username == "admin" || username == "user") && password == "12345")
            {
                MessageBox.Show("Welcome! " + username);
            }
            // Check if both username and password fields are left empty
            // Kullanıcı adı ve şifre alanlarının ikisinin birden boş bırakılıp bırakılmadığını kontrol ediyoruz
            else if (string.IsNullOrEmpty(username + password))
            {
                MessageBox.Show("Please enter a username and password");
            }
            // Check if the username field is left empty
            // Kullanıcı adı alanının boş bırakılıp bırakılmadığını kontrol ediyoruz
            else if (string.IsNullOrEmpty(username))
            {
                MessageBox.Show("Please enter a username");
            }
            // Check if the password field is left empty
            // Şifre alanının boş bırakılıp bırakılmadığını kontrol ediyoruz
            else if (string.IsNullOrEmpty(password))
            {
                MessageBox.Show("Please enter a password");
            }
            // Fallback for incorrect credentials
            // Hatalı bilgiler için genel uyarı durumu
            else
            {
                MessageBox.Show("Invalid Login" + Environment.NewLine + "Please Try Again");
            }
        }

        {

        }

        private void textBox2_TextChanged(object sender, EventArgs e)
        {

        }
    }
}
